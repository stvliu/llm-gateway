### 1.2 表结构定义

#### 1.2.1 用户表 (users)

```sql
CREATE TABLE users (
    id BIGINT PRIMARY KEY,
    email VARCHAR(128) NOT NULL UNIQUE,
    name VARCHAR(128),
    status VARCHAR(16) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

COMMENT ON TABLE users IS '用户表';
COMMENT ON COLUMN users.email IS '用户邮箱，唯一标识';
COMMENT ON COLUMN users.status IS '状态：active, inactive, suspended';
```

#### 1.2.2 团队表 (teams)

```sql
CREATE TABLE teams (
    id BIGINT PRIMARY KEY,
    name VARCHAR(128) NOT NULL,
    description VARCHAR(256),
    status VARCHAR(16) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

COMMENT ON TABLE teams IS '团队表';
```

#### 1.2.3 用户-团队关联表 (user_teams)

```sql
CREATE TABLE user_teams (
    user_id BIGINT NOT NULL REFERENCES users(id),
    team_id BIGINT NOT NULL REFERENCES teams(id),
    role VARCHAR(32) DEFAULT 'member',
    created_at TIMESTAMP DEFAULT NOW(),
    PRIMARY KEY (user_id, team_id)
);

COMMENT ON TABLE user_teams IS '用户-团队关联表';
COMMENT ON COLUMN user_teams.role IS '角色：owner, admin, member';
```

#### 1.2.4 供应商表 (providers)

```sql
CREATE TABLE providers (
    id BIGINT PRIMARY KEY,
    name VARCHAR(64) NOT NULL UNIQUE,
    display_name VARCHAR(128),
    description VARCHAR(256),
    status VARCHAR(16) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

COMMENT ON TABLE providers IS '大模型供应商表';
COMMENT ON COLUMN providers.name IS '供应商标识：volcengine, baidu, tencent, deepseek';
```

#### 1.2.5 产品表 (products)

```sql
CREATE TABLE products (
    id BIGINT PRIMARY KEY,
    provider_id BIGINT NOT NULL REFERENCES providers(id),
    name VARCHAR(128) NOT NULL,
    product_type VARCHAR(32) NOT NULL,
    models JSONB NOT NULL,
    endpoints JSONB NOT NULL,
    quota_limit BIGINT,
    status VARCHAR(16) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    UNIQUE(provider_id, name)
);

CREATE INDEX idx_products_provider ON products(provider_id);
CREATE INDEX idx_products_type ON products(product_type);

COMMENT ON TABLE products IS '产品表';
COMMENT ON COLUMN products.product_type IS '类型：pay_as_you_go, subscription_coding, subscription_token';
COMMENT ON COLUMN products.models IS '模型列表，如 ["doubao-seed-code", "doubao-seed-1.6"]';
COMMENT ON COLUMN products.endpoints IS '多协议端点，如 {"native": "...", "openai": "...", "anthropic": "..."}';
COMMENT ON COLUMN products.quota_limit IS '订阅产品的额度限制（Token 数）';
```

**endpoints JSONB 结构**：

```json
{
  "native": "https://ark.cn-beijing.volces.com/api/coding/v3",
  "openai": "https://ark.cn-beijing.volces.com/api/v3",
  "anthropic": "https://ark.cn-beijing.volces.com/api/anthropic"
}
```

**models JSONB 结构**：

```json
["doubao-seed-code", "doubao-seed-1.6", "doubao-pro-32k"]
```

#### 1.2.6 产品 API Key 表 (product_api_keys)

```sql
CREATE TABLE product_api_keys (
    id BIGINT PRIMARY KEY,
    product_id BIGINT NOT NULL REFERENCES products(id),
    name VARCHAR(128),
    api_key_encrypted TEXT NOT NULL,
    weight INT DEFAULT 1,
    priority INT DEFAULT 1,
    status VARCHAR(16) DEFAULT 'active',
    last_used_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_product_api_keys_product ON product_api_keys(product_id);
CREATE INDEX idx_product_api_keys_status ON product_api_keys(status);

COMMENT ON TABLE product_api_keys IS '供应商侧 API Key 表';
COMMENT ON COLUMN product_api_keys.api_key_encrypted IS '加密存储的供应商 API Key';
COMMENT ON COLUMN product_api_keys.weight IS '负载均衡权重';
COMMENT ON COLUMN product_api_keys.priority IS '故障转移优先级，数值越小优先级越高';
```

#### 1.2.7 用户 API Key 表 (user_api_keys)

```sql
CREATE TABLE user_api_keys (
    id BIGINT PRIMARY KEY,
    team_id BIGINT NOT NULL REFERENCES teams(id),
    owner_user_id BIGINT REFERENCES users(id),
    product_id BIGINT NOT NULL REFERENCES products(id),
    key_hash VARCHAR(128) NOT NULL,
    key_prefix VARCHAR(16) NOT NULL,
    name VARCHAR(128),
    models JSONB,
    quota_limit BIGINT,
    status VARCHAR(16) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE UNIQUE INDEX idx_user_api_keys_key_hash ON user_api_keys(key_hash);
CREATE INDEX idx_user_api_keys_team ON user_api_keys(team_id);
CREATE INDEX idx_user_api_keys_product ON user_api_keys(product_id);
CREATE INDEX idx_user_api_keys_status ON user_api_keys(status);

COMMENT ON TABLE user_api_keys IS '用户侧 API Key 表';
COMMENT ON COLUMN user_api_keys.key_hash IS 'Key 哈希值，用于认证';
COMMENT ON COLUMN user_api_keys.key_prefix IS 'Key 前缀，用于识别，如 sk-xxx';
COMMENT ON COLUMN user_api_keys.models IS '可访问的模型列表（子集），为空表示可访问产品全部模型';
COMMENT ON COLUMN user_api_keys.quota_limit IS 'Key 级别的额度限制';
```

#### 1.2.8 用量记录表 (usage_records)

```sql
CREATE TABLE usage_records (
    id BIGINT PRIMARY KEY,
    user_api_key_id BIGINT NOT NULL REFERENCES user_api_keys(id),
    team_id BIGINT NOT NULL REFERENCES teams(id),
    product_id BIGINT NOT NULL REFERENCES products(id),
    product_api_key_id BIGINT REFERENCES product_api_keys(id),
    model VARCHAR(64) NOT NULL,
    protocol VARCHAR(16) NOT NULL,
    input_tokens BIGINT,
    output_tokens BIGINT,
    cost DECIMAL(10,6),
    request_id VARCHAR(64),
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_usage_records_user_key ON usage_records(user_api_key_id);
CREATE INDEX idx_usage_records_team ON usage_records(team_id);
CREATE INDEX idx_usage_records_product ON usage_records(product_id);
CREATE INDEX idx_usage_records_created_at ON usage_records(created_at);

COMMENT ON TABLE usage_records IS '用量记录表';
```

## 2. 索引设计

### 2.1 主键与唯一索引

| 表 | 索引名 | 类型 | 字段 |
|---|--------|------|------|
| users | users_pkey | PRIMARY | id |
| users | users_email_key | UNIQUE | email |
| teams | teams_pkey | PRIMARY | id |
| providers | providers_pkey | PRIMARY | id |
| providers | providers_name_key | UNIQUE | name |
| products | products_pkey | PRIMARY | id |
| products | products_provider_id_name_key | UNIQUE | provider_id, name |
| user_api_keys | user_api_keys_pkey | PRIMARY | id |
| user_api_keys | idx_user_api_keys_key_hash | UNIQUE | key_hash |

### 2.2 外键索引

| 表 | 索引名 | 字段 | 用途 |
|---|--------|------|------|
| user_teams | - | user_id, team_id | 用户查询团队、团队查询用户 |
| products | idx_products_provider | provider_id | 查询供应商下产品 |
| product_api_keys | idx_product_api_keys_product | product_id | 查询产品的 API Key |
| user_api_keys | idx_user_api_keys_team | team_id | 查询团队的 API Key |
| user_api_keys | idx_user_api_keys_product | product_id | 查询产品被哪些 Key 使用 |
| usage_records | idx_usage_records_user_key | user_api_key_id | 用户用量统计 |
| usage_records | idx_usage_records_team | team_id | 团队用量统计 |
| usage_records | idx_usage_records_product | product_id | 产品用量统计 |

### 2.3 业务索引

| 表 | 索引名 | 字段 | 用途 |
|---|--------|------|------|
| products | idx_products_type | product_type | 按类型查询产品 |
| product_api_keys | idx_product_api_keys_status | status | 筛选有效 Key |
| user_api_keys | idx_user_api_keys_status | status | 筛选有效 Key |
| usage_records | idx_usage_records_created_at | created_at | 时间范围查询 |

## 4. 数据安全

### 4.1 密钥存储策略

| 密钥类型 | 存储方式 | 字段 |
|---------|---------|------|
| Product API Key | AES-256 加密 | `api_key_encrypted` |
| User API Key | SHA-256 哈希 | `key_hash` |

### 4.2 加密实现

**Product API Key 加密**：

```java
@Configuration
public class ApiKeyEncryptionConfig {

    @Value("${gateway.encryption.secret}")
    private String encryptionSecret;

    public String encrypt(String plainText) {
        // AES-256-GCM 加密
        byte[] key = deriveKey(encryptionSecret);
        Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
        cipher.init(Cipher.ENCRYPT_MODE, new SecretKeySpec(key, "AES"));
        byte[] encrypted = cipher.doFinal(plainText.getBytes(StandardCharsets.UTF_8));
        return Base64.getEncoder().encodeToString(encrypted);
    }

    public String decrypt(String encryptedText) {
        byte[] key = deriveKey(encryptionSecret);
        byte[] decoded = Base64.getDecoder().decode(encryptedText);
        Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
        cipher.init(Cipher.DECRYPT_MODE, new SecretKeySpec(key, "AES"));
        byte[] decrypted = cipher.doFinal(decoded);
        return new String(decrypted, StandardCharsets.UTF_8);
    }
}
```

**User API Key 哈希**：

```java
public class ApiKeyHasher {

    private static final int HASH_LENGTH = 32; // SHA-256

    public static String hash(String apiKey) {
        byte[] hash = MessageDigest.getInstance("SHA-256")
            .digest(apiKey.getBytes(StandardCharsets.UTF_8));
        return Hex.toHexString(hash);
    }

    public static boolean verify(String apiKey, String storedHash) {
        String computedHash = hash(apiKey);
        return MessageDigest.isEqual(
            computedHash.getBytes(StandardCharsets.UTF_8),
            storedHash.getBytes(StandardCharsets.UTF_8)
        );
    }
}
```

### 5.2 迁移脚本示例

**V1__init_schema.sql**：

```sql
-- 创建扩展插件
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 创建更新时间触发器函数
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 创建用户表
CREATE TABLE users (
    id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
    email VARCHAR(128) NOT NULL UNIQUE,
    name VARCHAR(128),
    status VARCHAR(16) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE TRIGGER users_updated_at
    BEFORE UPDATE ON users
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at();
```

## 6. 性能优化

### 6.1 分区策略

**usage_records 表按时间分区**：

```sql
CREATE TABLE usage_records (
    id BIGINT PRIMARY KEY,
    user_api_key_id BIGINT NOT NULL,
    team_id BIGINT NOT NULL,
    product_id BIGINT NOT NULL,
    product_api_key_id BIGINT,
    model VARCHAR(64) NOT NULL,
    protocol VARCHAR(16) NOT NULL,
    input_tokens BIGINT,
    output_tokens BIGINT,
    cost DECIMAL(10,6),
    request_id VARCHAR(64),
    created_at TIMESTAMP DEFAULT NOW()
) PARTITION BY RANGE (created_at);

-- 创建月度分区
CREATE TABLE usage_records_2026_05 PARTITION OF usage_records
    FOR VALUES FROM ('2026-05-01') TO ('2026-06-01');

CREATE TABLE usage_records_2026_06 PARTITION OF usage_records
    FOR VALUES FROM ('2026-06-01') TO ('2026-07-01');
```

### 6.3 查询优化建议

| 场景 | 优化策略 |
|------|---------|
| API Key 认证 | 使用 `key_hash` 唯一索引，O(1) 查询 |
| 产品模型查询 | JSONB `@>` 操作符利用 GIN 索引 |
| 用量统计 | 按时间分区 + 时间索引，减少扫描范围 |
| 热点产品 | 缓存产品配置到 Redis，减少数据库查询 |
